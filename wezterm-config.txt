local wezterm = require("wezterm")
local config = wezterm.config_builder()

-- 🖥️ INSTANCIA DE DOMINIO (WSL)
config.wsl_domains = {
  {
    name = 'WSL:Ubuntu',
    distribution = 'Ubuntu',
  },
}
config.default_domain = 'WSL:Ubuntu'

-- 🔤 CONFIGURACIÓN DE TIPOGRAFÍA (Estilo macOS Nativo)
config.font = wezterm.font("SFMono Nerd Font", { weight = "Regular" })
config.font_size = 14.0
config.line_height = 1.35  -- Espaciado idéntico a la imagen de Mac

-- 🛠️ OPTIMIZACIÓN DE RENDERIZADO PARA SÍMBOLOS (FREE TYPE)
config.freetype_load_target = "Light"
config.freetype_render_target = "HorizontalLcd"
config.allow_square_glyphs_to_overflow_width = "WhenFollowedBySpace"
config.adjust_window_size_when_changing_font_size = false

-- 🪟 CONFIGURACIÓN GEOMÉTRICA DE LA VENTANA
config.use_resize_increments = true  -- Elimina la franja muerta debajo de Neovim
config.window_padding = {
  left = 0,
  right = 0,
  top = 0,
  bottom = 0,
}

-- Estilo de ventana con botones superiores nativos e integrados
config.window_decorations = "INTEGRATED_BUTTONS | RESIZE"
-- config.integrated_title_button_alignment = "Left"
config.win32_system_backdrop = "Mica"

-- 📑 DISEÑO DE LAS PESTAÑAS (FANCY BAR - Estilo Mac Integrado)
config.enable_tab_bar = true
config.use_fancy_tab_bar = true
config.show_tabs_in_tab_bar = true
config.show_new_tab_button_in_tab_bar = true

config.window_frame = {
  font = wezterm.font("SFMono Nerd Font", { weight = "Bold" }),
  font_size = 11.0,
  -- Sincroniza el marco superior con el color gris de tu Neovim
  active_titlebar_bg = "#1c1c1c",
  inactive_titlebar_bg = "#161616",
}

-- 🎨 PALETA DE COLORES MONOCROMÁTICA INTEGRADA (SINCRONIZADA CON NEOVIM)
config.colors = {
  foreground = "#c0c0c0",
  background = "#1c1c1c",       -- El gris exacto de tu editor para fusión total
  cursor_bg = "#ffffff",        -- Cursor blanco minimalista clásico de Mac
  cursor_border = "#ffffff",
  cursor_fg = "#1c1c1c",
  selection_bg = "#343434",     -- Sombreado de selección gris oscuro sutil
  selection_fg = "#ffffff",
  
  -- Colores ANSI estándar para la terminal
  ansi = { "#1c1c1c", "#e06c75", "#98c379", "#d19a66", "#61afef", "#c678dd", "#56b6c2", "#abb2bf" },
  brights = { "#5c6370", "#e06c75", "#98c379", "#d19a66", "#61afef", "#c678dd", "#56b6c2", "#ffffff" },

  tab_bar = {
    -- Fondo de la barra superior (Lo igualamos al fondo de Neovim para ocultar la línea divisoria)
    background = "#1c1c1c",

    -- Pestaña Activa: Sutilmente más clara que el fondo para que se note cuál usas
    active_tab = {
      bg_color = "#282828",
      fg_color = "#ffffff",
      intensity = "Bold",
    },

    -- Pestañas Inactivas: Totalmente planas con el fondo, pero con letras más legibles
    inactive_tab = {
      bg_color = "#1c1c1c",
      fg_color = "#a0a0a0", -- Gris más claro que antes para leerlas sin esfuerzo
    },

    inactive_tab_hover = {
      bg_color = "#242424",
      fg_color = "#ffffff",
      italic = true,
    },

    new_tab = {
      bg_color = "#1c1c1c",
      fg_color = "#a0a0a0",
    },

    new_tab_hover = {
      bg_color = "#242424",
      fg_color = "#ffffff",
    },
  }
}

-- 🖱️ CONTROL DE MOUSE (Scroll Suave)
config.mouse_bindings = {
  {
    event = { Down = { streak = 1, button = { WheelUp = 1 } } },
    mods = "NONE",
    action = wezterm.action.ScrollByLine(-3),
    alt_screen = false,
  },
  {
    event = { Down = { streak = 1, button = { WheelDown = 1 } } },
    mods = "NONE",
    action = wezterm.action.ScrollByLine(3),
    alt_screen = false,
  },
}

-- ⌨️ ATAJOS DE TECLADO PERSONALIZADOS
config.use_dead_keys = false

config.leader = {
  key = "a",
  mods = "CTRL",
  timeout_milliseconds = 1000,
}

config.keys = {
  -- Pestaña Nueva
  {
    key = "t",
    mods = "ALT",
    action = wezterm.action.SpawnTab("CurrentPaneDomain"),
  },
  -- Ventana Nueva
  {
    key = "n",
    mods = "ALT",
    action = wezterm.action.SpawnWindow,
  },
  -- Navegación de pestañas
  {
    key = "[",
    mods = "ALT",
    action = wezterm.action.ActivateTabRelative(-1),
  },
  {
    key = "]",
    mods = "ALT",
    action = wezterm.action.ActivateTabRelative(1),
  },
  -- Cerrar pestaña
  {
    key = "w",
    mods = "ALT",
    action = wezterm.action.CloseCurrentTab { confirm = true },
  },
}

return config
